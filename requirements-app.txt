# SpatialMind Studio: the packaged application's runtime and build dependencies.
#
# requirements.txt includes optional workstation tools such as napari, vitessce,
# chromadb and celery, which are not on the Studio's active import path. The
# Studio includes Scanpy/Squidpy and their spatialdata/dask dependencies. Build
# from this smaller environment, not a workstation with deep-learning runtimes.
#
#   python3.11 -m venv .venv-app
#   .venv-app/bin/python -m pip install -r requirements-app.txt

-c packaging/constraints-macos.txt

-e .
setuptools<81

# Web layer
fastapi
uvicorn[standard]
pydantic

# to opening the default browser.
pywebview; sys_platform == "darwin"

# Numerics and tables
numpy<2
scipy
pandas
pyarrow
h5py

# Single-cell and spatial analysis
anndata
scanpy
squidpy
scikit-learn

# Clustering and graphs used by scanpy/squidpy
igraph
leidenalg

pynndescent

# Figures, morphology metadata, and reports
matplotlib

reportlab

python-docx
openpyxl
python-multipart

pyinstaller
